> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a Pairs section: a `/pairs` selector page where the user picks two players and navigates to `/pairs/:a/:b`, which shows the pair's combined stats, time-series charts, form summary, and rival/bogey/freebie pairs (equivalent of the player insights but in pairs context).

**Architecture:** Four new backend endpoints share a `_h2h_pair_stats` helper in `services/insights.py` that finds opponent pairs (by joining two aliased GamePlayer records for the opposing team) with ≥5 games faced as a specific pair. A new `get_pair_time_series` service function (in `services/stats.py`) mirrors `get_time_series` but filters to games where both players were on the same team. The frontend reuses `FormSummaryCard`, `WinRateChart`, `AvgPointsChart`, and `FormHeatmap` from Features 3–4. Two new pair-specific insight components handle the different schema. All stats respect the global player filter.

**Tech Stack:** FastAPI + SQLAlchemy (backend), React 19 + TypeScript + Tailwind v4 (frontend).

```python
def get_pair_time_series(
    db: Session,
    player_a_id: int,
    player_b_id: int,
    player_ids: list[int] | None = None,
) -> dict[str, Any]:
    for pid in (player_a_id, player_b_id):
        if not db.get(Player, pid):
            raise KeyError(f"Player {pid} not found")
    valid_ids = _valid_game_ids(player_ids)

    gpa = aliased(GamePlayer)
    gpb = aliased(GamePlayer)
    won_case = case(
        ((gpa.team == "A") & (Game.team_a_score > Game.team_b_score), 1),
        ((gpa.team == "B") & (Game.team_b_score > Game.team_a_score), 1),
        else_=0,
    )
    points_for = case((gpa.team == "A", Game.team_a_score), else_=Game.team_b_score)
    points_against = case((gpa.team == "A", Game.team_b_score), else_=Game.team_a_score)

    q = (
        db.query(
            _session_rank.c.session,
            Game.played_on,
            Game.game_number,
            won_case.label("won"),
            points_for.label("score_for"),
            points_against.label("score_against"),
        )
        .join(gpa, (gpa.game_id == Game.id) & (gpa.player_id == player_a_id))
        .join(gpb, (gpb.game_id == Game.id) & (gpb.player_id == player_b_id) & (gpb.team == gpa.team))
        .join(_session_rank, _session_rank.c.played_on == Game.played_on)
    )
    if valid_ids is not None:
        q = q.filter(Game.id.in_(valid_ids))

    rows = q.order_by(_session_rank.c.session, Game.game_number).all()

    sessions = []
    for session_num, group in groupby(rows, key=lambda r: r.session):
        game_rows = list(group)
        wins = sum(r.won for r in game_rows)
        games_played = len(game_rows)
        avg_pf = sum(r.score_for for r in game_rows) / games_played
        avg_pa = sum(r.score_against for r in game_rows) / games_played
        sessions.append({
            "session": session_num,
            "played_on": game_rows[0].played_on,
            "games_played": games_played,
            "wins": wins,
            "losses": games_played - wins,
            "win_rate": round(wins / games_played, 4) if games_played else 0.0,
            "avg_points_for": round(avg_pf, 2),
            "avg_points_against": round(avg_pa, 2),
            "games": [
                {
                    "game_number": r.game_number,
                    "won": bool(r.won),
                    "score_for": r.score_for,
                    "score_against": r.score_against,
                }
                for r in game_rows
            ],
        })

    return {"player_a_id": player_a_id, "player_b_id": player_b_id, "sessions": sessions}
```

```python
def _h2h_pair_stats(
    db: Session,
    player_a_id: int,
    player_b_id: int,
    player_ids: list[int] | None = None,
) -> list[dict[str, Any]]:
    """For all opponent pairs who faced (player_a, player_b) >= MIN_H2H_GAMES times,
    return per-opponent-pair stats. Both h2h and overall win rates respect player_ids."""
    valid_ids = _valid_game_ids(player_ids)

    gpa = aliased(GamePlayer)   # player_a in target pair
    gpb = aliased(GamePlayer)   # player_b in target pair
    opp1 = aliased(GamePlayer)  # opponent with smaller player_id
    opp2 = aliased(GamePlayer)  # opponent with larger player_id

    opp_won = case(
        ((opp1.team == "A") & (Game.team_a_score > Game.team_b_score), 1),
        ((opp1.team == "B") & (Game.team_b_score > Game.team_a_score), 1),
        else_=0,
    )
    score_diff = func.abs(Game.team_a_score - Game.team_b_score)

    h2h_q = (
        db.query(
            opp1.player_id.label("opp_a_id"),
            opp2.player_id.label("opp_b_id"),
            func.count(Game.id).label("games"),
            func.avg(score_diff).label("avg_score_diff"),
            func.avg(opp_won).label("opp_win_rate_vs"),
        )
        .join(gpa, (gpa.game_id == Game.id) & (gpa.player_id == player_a_id))
        .join(gpb, (gpb.game_id == Game.id) & (gpb.player_id == player_b_id) & (gpb.team == gpa.team))
        .join(opp1, (opp1.game_id == Game.id) & (opp1.team != gpa.team))
        .join(opp2, (opp2.game_id == Game.id) & (opp2.team == opp1.team) & (opp2.player_id > opp1.player_id))
    )
    if valid_ids is not None:
        h2h_q = h2h_q.filter(Game.id.in_(valid_ids))

    h2h_rows = (
        h2h_q
        .group_by(opp1.player_id, opp2.player_id)
        .having(func.count(Game.id) >= MIN_H2H_GAMES)
        .all()
    )
    if not h2h_rows:
        return []

    # Overall win rate for each opponent pair across all filtered games
    opp_pairs = [(r.opp_a_id, r.opp_b_id) for r in h2h_rows]
    gpc = aliased(GamePlayer)
    gpd = aliased(GamePlayer)
    cd_won = case(
        ((gpc.team == "A") & (Game.team_a_score > Game.team_b_score), 1),
        ((gpc.team == "B") & (Game.team_b_score > Game.team_a_score), 1),
        else_=0,
    )
    overall_q = (
        db.query(
            gpc.player_id.label("opp_a_id"),
            gpd.player_id.label("opp_b_id"),
            func.avg(cd_won).label("overall_win_rate"),
        )
        .join(Game, gpc.game_id == Game.id)
        .join(gpd, (gpd.game_id == gpc.game_id) & (gpd.team == gpc.team) & (gpd.player_id > gpc.player_id))
        .filter(or_(*[and_(gpc.player_id == a, gpd.player_id == b) for a, b in opp_pairs]))
    )
    if valid_ids is not None:
        overall_q = overall_q.filter(Game.id.in_(valid_ids))
    overall_map = {
        (r.opp_a_id, r.opp_b_id): round(float(r.overall_win_rate or 0), 4)
        for r in overall_q.group_by(gpc.player_id, gpd.player_id).all()
    }

    # Batch name lookup
    all_opp_ids = {r.opp_a_id for r in h2h_rows} | {r.opp_b_id for r in h2h_rows}
    names = {p.id: p.canonical_name for p in db.query(Player).filter(Player.id.in_(all_opp_ids)).all()}

    results = []
    for r in h2h_rows:
        owr = overall_map.get((r.opp_a_id, r.opp_b_id), 0.0)
        h2h_wr = round(float(r.opp_win_rate_vs or 0), 4)
        results.append({
            "player_a_id": r.opp_a_id,
            "player_b_id": r.opp_b_id,
            "canonical_name_a": names.get(r.opp_a_id, ""),
            "canonical_name_b": names.get(r.opp_b_id, ""),
            "games": r.games,
            "avg_score_diff": round(float(r.avg_score_diff or 0), 2),
            "win_rate_vs_you": h2h_wr,
            "overall_win_rate": owr,
            "delta": round(h2h_wr - owr, 4),
        })
    return results


def get_pair_rivals(
    db: Session, player_a_id: int, player_b_id: int, player_ids: list[int] | None = None
) -> list[dict[str, Any]]:
    for pid in (player_a_id, player_b_id):
        if not db.get(Player, pid):
            raise KeyError(f"Player {pid} not found")
    entries = _h2h_pair_stats(db, player_a_id, player_b_id, player_ids)
    return sorted(entries, key=lambda e: e["avg_score_diff"])[:TOP_N]


def get_pair_bogeys(
    db: Session, player_a_id: int, player_b_id: int, player_ids: list[int] | None = None
) -> list[dict[str, Any]]:
    for pid in (player_a_id, player_b_id):
        if not db.get(Player, pid):
            raise KeyError(f"Player {pid} not found")
    entries = _h2h_pair_stats(db, player_a_id, player_b_id, player_ids)
    return sorted(entries, key=lambda e: e["delta"], reverse=True)[:TOP_N]


def get_pair_freebies(
    db: Session, player_a_id: int, player_b_id: int, player_ids: list[int] | None = None
) -> list[dict[str, Any]]:
    for pid in (player_a_id, player_b_id):
        if not db.get(Player, pid):
            raise KeyError(f"Player {pid} not found")
    entries = _h2h_pair_stats(db, player_a_id, player_b_id, player_ids)
    return sorted(entries, key=lambda e: e["delta"])[:TOP_N]
```

```typescript
import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card'
import type { PairInsightEntry } from '../api/stats'

interface Props {
  title: string
  entries: PairInsightEntry[]
}

export default function PairInsightCard({ title, entries }: Props) {
  const subtitle =
    title === 'Bogeys' ? 'Overperform against you' : 'Underperform against you'

  return (
    <Card>
      <CardHeader>
        <CardTitle className="text-base">{title}</CardTitle>
        <p className="text-xs text-muted-foreground">{subtitle} (min 5 games)</p>
      </CardHeader>
      <CardContent>
        {entries.length === 0 ? (
          <p className="text-sm text-muted-foreground">Not enough data</p>
        ) : (
          <ol className="space-y-1.5">
            {entries.map((e, i) => {
              const vsYou = `${Math.round(e.win_rate_vs_you * 100)}%`
              const usual = `${Math.round(e.overall_win_rate * 100)}%`
              const deltaPp = Math.round(Math.abs(e.delta) * 100)
              const colour = e.delta > 0 ? 'text-red-400' : 'text-green-400'
              const sign = e.delta > 0 ? '+' : '−'
              return (
                <li key={`${e.player_a_id}-${e.player_b_id}`} className="flex items-center justify-between text-sm">
                  <span className="flex items-center gap-2">
                    <span className="w-4 text-muted-foreground">{i + 1}.</span>
                    <span className="font-medium">
                      {e.canonical_name_a} &amp; {e.canonical_name_b}
                    </span>
                  </span>
                  <span className="text-right text-muted-foreground">
                    {vsYou} vs you · {usual} usual{' '}
                    <span className={`font-medium ${colour}`}>
                      {sign}{deltaPp}pp
                    </span>
                  </span>
                </li>
              )
            })}
          </ol>
        )}
      </CardContent>
    </Card>
  )
}
```

```typescript
import { useState, useEffect } from 'react'
import { useNavigate } from 'react-router-dom'
import { usePlayerFilter } from '../context/PlayerFilterContext'
import { getPlayerPartnerships } from '../api/players'
import { Button } from '@/components/ui/button'
import type { PlayerPartnership } from '../types'

export default function PairsPage() {
  const { allPlayers, selectedIds } = usePlayerFilter()
  const navigate = useNavigate()

  const filteredPlayers = allPlayers.filter((p) => selectedIds.includes(p.id))

  const [playerAId, setPlayerAId] = useState<number | ''>('')
  const [playerBId, setPlayerBId] = useState<number | ''>('')
  const [partners, setPartners] = useState<PlayerPartnership[]>([])

  // When player A changes, fetch their partners and reset player B
  useEffect(() => {
    setPlayerBId('')
    setPartners([])
    if (!playerAId) return
    getPlayerPartnerships(playerAId, selectedIds).then(setPartners).catch(() => setPartners([]))
  }, [playerAId, selectedIds])

  // Valid player B options: partners of player A who are also in the filtered set
  const partnerIds = new Set(partners.map((p) => p.partner_id))
  const playerBOptions = filteredPlayers.filter(
    (p) => p.id !== playerAId && partnerIds.has(p.id),
  )

  const canView = playerAId !== '' && playerBId !== ''

  const handleView = () => {
    if (!canView) return
    const a = Math.min(Number(playerAId), Number(playerBId))
    const b = Math.max(Number(playerAId), Number(playerBId))
    navigate(`/pairs/${a}/${b}`)
  }

  return (
    <div className="p-6">
      <h1 className="mb-6 text-xl font-semibold">Pairs</h1>
      <div className="flex flex-col gap-4 sm:flex-row sm:items-end">
        <div className="flex flex-col gap-1">
          <label htmlFor="player-a" className="text-sm font-medium">
            Player A
          </label>
          <select
            id="player-a"
            value={playerAId}
            onChange={(e) => setPlayerAId(e.target.value ? Number(e.target.value) : '')}
            className="rounded border border-border bg-background px-3 py-1.5 text-sm"
          >
            <option value="">Select a player…</option>
            {filteredPlayers.map((p) => (
              <option key={p.id} value={p.id}>
                {p.canonical_name}
              </option>
            ))}
          </select>
        </div>

        <div className="flex flex-col gap-1">
          <label htmlFor="player-b" className="text-sm font-medium">
            Player B
          </label>
          <select
            id="player-b"
            value={playerBId}
            onChange={(e) => setPlayerBId(e.target.value ? Number(e.target.value) : '')}
            disabled={!playerAId || playerBOptions.length === 0}
            className="rounded border border-border bg-background px-3 py-1.5 text-sm disabled:opacity-50"
          >
            <option value="">
              {!playerAId
                ? 'Select Player A first'
                : playerBOptions.length === 0
                  ? 'No partners found'
                  : 'Select a partner…'}
            </option>
            {playerBOptions.map((p) => (
              <option key={p.id} value={p.id}>
                {p.canonical_name}
              </option>
            ))}
          </select>
        </div>

        <Button onClick={handleView} disabled={!canView}>
          View Pair
        </Button>
      </div>
    </div>
  )
```

- [ ] **Step 2: Create `frontend/src/pages/PairDetailPage.tsx`**

```typescript
import { useEffect, useState } from 'react'
import { useParams, Link } from 'react-router-dom'
import { usePlayerFilter } from '../context/PlayerFilterContext'
import {
  getPairTimeSeries,
  getPairRivals,
  getPairBogeys,
  getPairFreebies,
} from '../api/stats'
import type { PairTimeSeriesResponse, PairRivalEntry, PairInsightEntry } from '../api/stats'
import StatCard from '../components/StatCard'
import FormSummaryCard from '../components/FormSummaryCard'
import WinRateChart from '../components/WinRateChart'
import AvgPointsChart from '../components/AvgPointsChart'
import FormHeatmap from '../components/FormHeatmap'
import PairRivalsCard from '../components/PairRivalsCard'
import PairInsightCard from '../components/PairInsightCard'
import { usePlayerNames } from '../hooks/usePlayerNames'

export default function PairDetailPage() {
  const { a, b } = useParams<{ a: string; b: string }>()
  // Normalise so smaller id is always a
  const playerAId = Math.min(Number(a), Number(b))
  const playerBId = Math.max(Number(a), Number(b))

  const { selectedIds, allPlayers } = usePlayerFilter()

  const playerA = allPlayers.find((p) => p.id === playerAId)
  const playerB = allPlayers.find((p) => p.id === playerBId)
  const pairLabel =
    playerA && playerB
      ? `${playerA.canonical_name} & ${playerB.canonical_name}`
      : `Player ${playerAId} & ${playerBId}`

  const [timeSeries, setTimeSeries] = useState<PairTimeSeriesResponse | null>(null)
  const [rivals, setRivals] = useState<PairRivalEntry[]>([])
  const [bogeys, setBogeys] = useState<PairInsightEntry[]>([])
  const [freebies, setFreebies] = useState<PairInsightEntry[]>([])
  const [error, setError] = useState<string | null>(null)
  const [loading, setLoading] = useState(true)

  useEffect(() => {
    setLoading(true)
    setError(null)
    Promise.all([
      getPairTimeSeries(playerAId, playerBId, selectedIds),
      getPairRivals(playerAId, playerBId, selectedIds),
      getPairBogeys(playerAId, playerBId, selectedIds),
      getPairFreebies(playerAId, playerBId, selectedIds),
    ])
      .then(([ts, rv, bg, fb]) => {
        setTimeSeries(ts)
        setRivals(rv)
        setBogeys(bg)
        setFreebies(fb)
      })
      .catch((e: Error) => setError(e.message))
      .finally(() => setLoading(false))
  }, [playerAId, playerBId, selectedIds])

  if (error) return <p className="p-6 text-destructive">{error}</p>
  if (loading && !timeSeries) return <p className="p-6 text-muted-foreground">Loading…</p>

  const sessions = timeSeries?.sessions ?? []
  const totalGames = sessions.reduce((s, x) => s + x.games_played, 0)
  const totalWins = sessions.reduce((s, x) => s + x.wins, 0)
  const totalLosses = totalGames - totalWins
  const winRate = totalGames > 0 ? totalWins / totalGames : 0
  const avgPoints =
    totalGames > 0
      ? sessions.reduce((s, x) => s + x.avg_points_for * x.games_played, 0) / totalGames
      : 0

  return (
    <div className={`p-6 space-y-6 transition-opacity ${loading ? 'opacity-50' : ''}`}>
      <div>
        <div className="mb-1 text-sm text-muted-foreground">
          <Link to="/pairs" className="hover:text-yellow-400">Pairs</Link>
          {' / '}
          {pairLabel}
        </div>
        <h1 className="text-2xl font-bold">{pairLabel}</h1>
      </div>

      {/* Stat cards */}
      <div className="grid grid-cols-2 gap-3 sm:grid-cols-5">
        <StatCard label="Games" value={totalGames} />
        <StatCard label="Wins" value={totalWins} />
        <StatCard label="Losses" value={totalLosses} />
        <StatCard label="Win Rate" value={`${(winRate * 100).toFixed(1)}%`} />
        <StatCard label="Avg Pts" value={avgPoints.toFixed(1)} />
      </div>

      {/* Form + insights */}
      <div className="grid grid-cols-1 gap-4 md:grid-cols-2">
        <FormSummaryCard sessions={sessions} />
        <PairRivalsCard rivals={rivals} />
        <PairInsightCard title="Bogeys" entries={bogeys} />
        <PairInsightCard title="Freebies" entries={freebies} />
      </div>

      {/* Charts */}
      <div className="grid grid-cols-1 gap-4 lg:grid-cols-2">
        <div className="rounded-lg border border-border p-4">
          <WinRateChart sessions={sessions} />
        </div>
        <div className="rounded-lg border border-border p-4">
          <AvgPointsChart sessions={sessions} />
        </div>
        <div className="rounded-lg border border-border p-4 lg:col-span-2">
          <FormHeatmap sessions={sessions} />
        </div>
      </div>
    </div>
  )
}
```
